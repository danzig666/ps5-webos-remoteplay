cmake_minimum_required(VERSION 3.20)
project(webos_m0 C)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Use scripts/build-webos.sh with the webOS SDK")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.24)
set(OPUS_FOUND TRUE)
set(OPUS_INCLUDE_DIRS "${CMAKE_SOURCE_DIR}/build/deps-root/usr/include/opus")
set(OPUS_LIBRARIES "${CMAKE_SOURCE_DIR}/build/deps-root/usr/lib/libopus.a")
if(NOT EXISTS "${OPUS_LIBRARIES}")
  message(FATAL_ERROR "Run scripts/build-deps.sh for pinned Opus (NDL backend dependency)")
endif()
include(cmake/SS4SM0.cmake)
# Link app calls to the private SDL2 compatibility SONAME. NDL loads Aurora's shim.
add_library(sdl2-webos SHARED IMPORTED)
set_target_properties(sdl2-webos PROPERTIES IMPORTED_LOCATION "${CMAKE_SOURCE_DIR}/build/sdl-runtime/lib/libSDL2-webos.so.0")
add_library(sdl2_ndl_shim SHARED upstream/aurora-tv/src/app/platform/webos/sdl2_ndl_shim.c)
target_link_libraries(sdl2_ndl_shim PRIVATE dl)
set_target_properties(sdl2_ndl_shim PROPERTIES OUTPUT_NAME "SDL2-2.0" SUFFIX ".so.0" NO_SONAME TRUE INSTALL_RPATH "$ORIGIN")
target_link_options(sdl2_ndl_shim PRIVATE "-Wl,-soname,libSDL2-2.0.so.0")
install(TARGETS sdl2_ndl_shim LIBRARY DESTINATION lib)
add_executable(m0 platform/webos/main.c platform/webos/media_bridge.c platform/webos/input_bridge.c platform/webos/device_probe.c)
target_compile_options(m0 PRIVATE -Wall -Wextra -Werror)
target_link_libraries(m0 PRIVATE "${CMAKE_SOURCE_DIR}/target/armv7-unknown-linux-gnueabi/release/libwebos_m0.a" ss4s SDL2_test sdl2-webos pthread dl m atomic)
set_target_properties(m0 PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
install(TARGETS m0 RUNTIME DESTINATION . COMPONENT M0)

target_include_directories(m0 PRIVATE ${SDL2_INCLUDE_DIRS})

option(BUILD_CLIENT "Build direct Remote Play client" OFF)
if(BUILD_CLIENT)
  add_executable(ps5remote platform/client/main.c platform/client/platform.c platform/client/input.c platform/client/vendor/dualsense_usb.c platform/webos/device_probe.c)
  target_include_directories(ps5remote PRIVATE platform/client/compat platform/client/vendor upstream/aurora-tv/core/moonlight-common-c/src ${SDL2_INCLUDE_DIRS} ${OPUS_INCLUDE_DIRS})
  target_compile_options(ps5remote PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(ps5remote PRIVATE "${CMAKE_SOURCE_DIR}/target/armv7-unknown-linux-gnueabi/release/libps5remote_app.a" ss4s SDL2_test sdl2-webos ${OPUS_LIBRARIES} pthread dl m atomic)
  set_target_properties(ps5remote PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
  install(TARGETS ps5remote RUNTIME DESTINATION . COMPONENT Client)
endif()
